{{- define "RenderImageSimple" -}} {{- $imgObj := .imgObj -}} {{- $src := .src -}} {{- $alt := .alt -}} {{- end -}} {{- define "RenderImageResponsive" -}} {{- $imgObj := .imgObj -}} {{- $alt := .alt -}} {{- $w := $imgObj.Width -}} {{- $targets := slice 400 800 1200 -}} {{- $variants := slice -}} {{- range $targets -}} {{- if le . $w -}} {{- $variants = $variants | append ($imgObj.Resize (printf "%dx webp" .)) -}} {{- end -}} {{- end -}} {{- if and $variants (gt (len $targets) (len $variants)) -}} {{- if gt $w (index $variants (sub (len $variants) 1)).Width -}} {{- $variants = $variants | append ($imgObj.Resize (printf "%dx webp" $w)) -}} {{- end -}} {{- end -}} {{- if not $variants -}} {{- $variants = slice ($imgObj.Resize (printf "%dx webp" $w)) -}} {{- end -}} {{- $srcset := slice -}} {{- range $variants -}} {{- $srcset = $srcset | append (printf "%s %dw" .RelPermalink .Width) -}} {{- end -}} {{- $zoom := index $variants (sub (len $variants) 1) -}} {{- if gt $w 1920 -}} {{- $zoom = $imgObj.Resize "1920x webp" -}} {{- end -}} {{- end -}} {{- define "RenderImageCaption" -}} {{- with .caption -}} {{- end -}} {{- end -}} {{- $disableImageOptimizationMD := .Page.Site.Params.disableImageOptimizationMD | default false -}} {{- $urlStr := .Destination | safeURL -}} {{- $url := urls.Parse $urlStr -}} {{- $altText := .Text -}} {{- $caption := .Title -}} {{- $isRemote := findRE "^(https?|data)" $url.Scheme -}} {{- $resource := "" -}} {{- if not $isRemote -}} {{- $rel := $urlStr | strings.TrimPrefix "./" -}} {{- $ctx := $.Page -}} {{- range seq 6 -}} {{- if hasPrefix $rel "../" -}} {{- $rel = strings.TrimPrefix "../" $rel -}} {{- with $ctx.Parent }}{{ $ctx = . }}{{ end -}} {{- end -}} {{- end -}} {{- $resource = or ($ctx.Resources.GetMatch $rel) ($.Page.Resources.GetMatch $urlStr) (resources.Get $urlStr) -}} {{- end -}}
