"-") "date" $r.Date "stage" ($r.Params.stage | default "stable") "summary" ($r.Params.summary | default "") "content" $r.Content "releaseURL" $releaseURL "announcement" $announcement "downloadURL" ($r.Params.download_url | default "") ) -}}
