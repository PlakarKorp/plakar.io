{{- /* Tables on integration pages: horizontal scroll on small screens, comparison styling. A cell starting with "✓ ", "✗ " or "◐ " gets a green check, a red cross or an orange partial mark, followed by the rest of its text. The column whose header is "Plakar" is highlighted. */ -}} {{- $marks := slice (dict "sym" "✓" "class" "yes" "label" "Yes" "svg" ``) (dict "sym" "✗" "class" "no" "label" "No" "svg" ``) (dict "sym" "◐" "class" "partial" "label" "Partly" "svg" ``) -}} {{- $plakarCol := -1 -}} {{- range .THead -}} {{- range $i, $c := . -}} {{- if eq (strings.TrimSpace (plainify $c.Text)) "Plakar" }}{{ $plakarCol = $i }}{{ end -}} {{- end -}} {{- end -}}
{{- range .THead }} {{- range $i, $c := . }} {{ $c.Text }} {{- end }} {{- end }} {{- range .TBody }} {{- range $i, $c := . }} {{- if eq $i 0 }} {{ $c.Text }} {{- else }} {{- $text := string $c.Text }} {{- $cell := $text | safeHTML }} {{- range $marks }} {{- if hasPrefix $text .sym }} {{- $rest := strings.TrimSpace (strings.TrimPrefix .sym $text) }} {{- $cell = printf `%s%s%s` .class .svg .label (cond (ne $rest "") (printf `%s` $rest) "") | safeHTML }} {{- end }} {{- end }} {{ $cell }} {{- end }} {{- end }} {{- end }}
